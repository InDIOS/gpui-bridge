# Synced checkbox: the host keeps the state, so a second click unchecks it.
> {"op":"simulate","id":5,"event":"change"}
< {"type":"event","id":5,"event":"change","checked":true}
> {"op":"simulate","id":5,"event":"change"}
< {"type":"event","id":5,"event":"change","checked":false}
# Controlled checkbox: the host only reports; the state changes when the client sets it.

< {"type":"event","id":6,"event":"change","checked":true}
> {"op":"simulate","id":6,"event":"change"}
< {"type":"event","id":6,"event":"change","checked":true}
> {"op":"setProps","id":6,"props":{"checked":true}}
> {"op":"commit"}
> {"op":"simulate","id":6,"event":"change"}
< {"type":"event","id":6,"event":"change","checked":false}
