# Section 29: icons the client registers (`icons`) and uses in any `icon` prop.
> {"op":"hello","protocol":1}
< {"type":"ready","protocol":1}
# A name not registered (nor bundled) is invalid_props, as before.
> {"op":"create","id":1,"type":"Icon","props":{"name":"conformance-logo"}}
> {"op":"commit"}
< {"type":"error","code":"invalid_props","op":"create"}
# Registered first in the same batch, the name works in an Icon and in an `icon` prop.
> {"op":"icons","icons":{"conformance-logo":"<svg xmlns=\"http://www.w3.org/2000/svg\" viewBox=\"0 0 24 24\"><circle cx=\"12\" cy=\"12\" r=\"10\" fill=\"currentColor\"/></svg>"}}
> {"op":"create","id":1,"type":"Icon","props":{"name":"conformance-logo"}}
> {"op":"create","id":2,"type":"Badge","props":{"icon":"conformance-logo","label":"New"}}
> {"op":"commit"}
# Registering a name again replaces it.
> {"op":"icons","icons":{"conformance-logo":"<svg xmlns=\"http://www.w3.org/2000/svg\" viewBox=\"0 0 24 24\"><rect width=\"20\" height=\"20\" x=\"2\" y=\"2\"/></svg>"}}
> {"op":"commit"}

> {"op":"commit"}
< {"type":"error","code":"invalid_operation","op":"icons"}
